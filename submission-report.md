## Overview

`myapp` does the address arithmetic CIDR notation hides and the longest prefix match a router runs on every packet.

```bash
make all
./build/release/myapp info 132.178.227.11/25
./build/release/myapp split 192.168.1.0/24 4
./build/release/myapp lookup tables/example.txt 132.178.227.50 8.8.8.8 10.1.2.5 10.1.2.200
```

Exit codes: 0 on success (and with no arguments, which prints the usage), 1 for a bad command line or malformed address, 2 when the table file cannot be opened or has a bad line, 3 when an address matches no route. A lookup run with both a malformed and an unreachable address exits 1.

## Design

1. **Parsing.** `ip_parse` and `cidr_parse` share a hand-written scanner that reads a bounded number of decimal digits and refuses a leading zero, then insist on `.`, `/` or `'\0'` right after. Signs, spaces, octal-looking octets and trailing junk are rejected rather than guessed at.
2. **/0, /31 and /32.** `cidr_mask(0)` returns 0 before it ever shifts, since `0xFFFFFFFF << 32` is undefined behavior and on x86 would make the default route match only 0.0.0.0. Host counts and subnet offsets are computed in 64 bits so a /0 gives 4294967294 instead of overflowing. For /31 (RFC 3021) and /32 the first and last host are the network and broadcast addresses themselves, and the host count is 2 and 1 instead of the formula.
3. **Forwarding table.** `fwd_table_t` is an array of `route_t` that doubles with `realloc`. `table_add` copies the route and refuses host bits and duplicate blocks. `table_lookup` scans every route and keeps the containing one with the longest prefix. It never stops at the first match, so insertion order does not matter, and the default route wins only when nothing more specific matches.

`tests/lab-test.c` covers every function, including the boundary prefixes, each kind of malformed input, nested routes added in several orders, a table with no default route and a file with a bad line in the middle. `src/lab.c` has 100% line and branch coverage. The only exclusion is the `realloc` failure path.

## Experience

Once the mask was right, each subnet function was a line or two, so most of the work was in the edge cases. The /0 mask was the trap: the obvious shift is undefined behavior, and on x86 it seems to work while quietly breaking the default route. #include "lab.h"
#include <stdlib.h>
#include <string.h>

/** A forwarding table: a growable array of routes, scanned in full on lookup. */
struct fwd_table {
  route_t *routes; /**< the routes, in the order they were added */
  size_t count;    /**< how many routes are in use */
  size_t capacity; /**< how many routes fit before the next realloc() */
};

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

static bool is_digit(char c) { return c >= '0' && c <= '9'; }

/**
 * Parse an unsigned decimal number of at most max_digits digits with no
 * leading zero ("0" itself is fine). On success *value holds the number and
 * the returned pointer is the first character after it; NULL on failure.
 */
static const char *parse_decimal(const char *p, int max_digits, uint32_t *value)
{
  if (!is_digit(*p)) {
    return NULL;
  }
  if (p[0] == '0' && is_digit(p[1])) {
    return NULL;
  }
  uint32_t v = 0;
  int digits = 0;
  while (is_digit(*p)) {
    if (++digits > max_digits) {
      return NULL;
    }
    v = v * 10 + (uint32_t)(*p - '0');
    p++;
  }
  *value = v;
  return p;
}

/**
 * Parse a dotted quad at the start of text. Returns a pointer just past it,
 * or NULL if text does not start with a valid one.
 */
static const char *parse_quad(const char *text, uint32_t *addr)
{
  const char *p = text;
  uint32_t result = 0;
  for (int i = 0; i < 4; i++) {
    if (i > 0) {
      if (*p != '.') {
        return NULL;
      }
      p++;
    }
    uint32_t octet;
    p = parse_decimal(p, 3, &octet);
    if (p == NULL || octet > 255) {
      return NULL;
    }
    result = (result << 8) | octet;
  }
  *addr = result;
  return p;
}

bool ip_parse(const char *text, uint32_t *addr)
{
  if (text == NULL || addr == NULL) {
    return false;
  }
  uint32_t value;
  const char *end = parse_quad(text, &value);
  if (end == NULL || *end != '\0') {
    return false;
  }
  *addr = value;
  return true;
}

void ip_format(uint32_t addr, char out[IP_STR_LEN])
{
  snprintf(out, IP_STR_LEN, "%u.%u.%u.%u", (unsigned)(addr >> 24) & 0xFFu,
           (unsigned)(addr >> 16) & 0xFFu, (unsigned)(addr >> 8) & 0xFFu,
           (unsigned)addr & 0xFFu);
}

bool cidr_parse(const char *text, cidr_t *block)
{
  if (text == NULL || block == NULL) {
    return false;
  }
  uint32_t addr;
  const char *p = parse_quad(text, &addr);
  if (p == NULL || *p != '/') {
    return false;
  }
  uint32_t prefix;
  p = parse_decimal(p + 1, 2, &prefix);
  if (p == NULL || *p != '\0' || prefix > 32) {
    return false;
  }
  block->addr = addr;
  block->prefix = (uint8_t)prefix;
  return true;
}

void cidr_format(cidr_t block, char out[CIDR_STR_LEN])
{
  char ip[IP_STR_LEN];
  ip_format(block.addr, ip);
  unsigned prefix = block.prefix > 32 ? 32u : block.prefix;
  snprintf(out, CIDR_STR_LEN, "%s/%u", ip, prefix);
}

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

/** The prefix clamped to 32, the way cidr_mask() reads it. */
static unsigned prefix_of(cidr_t block)
{
  return block.prefix > 32 ? 32u : block.prefix;
}

uint32_t cidr_mask(uint8_t prefix)
{
  if (prefix == 0) {
    return 0; // 0xFFFFFFFF << 32 is undefined behavior, not 0
  }
  if (prefix > 32) {
    prefix = 32;
  }
  return UINT32_MAX << (32u - prefix);
}

uint32_t cidr_network(cidr_t block)
{
  return block.addr & cidr_mask(block.prefix);
}

uint32_t cidr_broadcast(cidr_t block)
{
  return block.addr | ~cidr_mask(block.prefix);
}

uint32_t cidr_first_host(cidr_t block)
{
  uint32_t network = cidr_network(block);
  return prefix_of(block) >= 31 ? network : network + 1;
}

uint32_t cidr_last_host(cidr_t block)
{
  uint32_t broadcast = cidr_broadcast(block);
  return prefix_of(block) >= 31 ? broadcast : broadcast - 1;
}

uint64_t cidr_host_count(cidr_t block)
{
  unsigned prefix = prefix_of(block);
  if (prefix == 32) {
    return 1;
  }
  if (prefix == 31) {
    return 2;
  }
  return ((uint64_t)1 << (32u - prefix)) - 2;
}

bool cidr_contains(cidr_t block, uint32_t addr)
{
  uint32_t mask = cidr_mask(block.prefix);
  return (addr & mask) == (block.addr & mask);
}

bool cidr_is_network(cidr_t block)
{
  return (block.addr & ~cidr_mask(block.prefix)) == 0;
}

bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out)
{
  if (out == NULL || !cidr_is_network(block)) {
    return false;
  }
  if (n == 0 || (n & (n - 1)) != 0) {
    return false;
  }
  unsigned borrowed = 0;
  while ((1u << borrowed) < n) {
    borrowed++;
  }
  unsigned prefix = prefix_of(block) + borrowed;
  if (prefix > 32 || i >= n) {
    return false;
  }
  // Shift in 64 bits: when n is 1 and the block is a /0, the shift is 32.
  uint64_t offset = (uint64_t)i << (32u - prefix);
  out->addr = cidr_network(block) + (uint32_t)offset;
  out->prefix = (uint8_t)prefix;
  return true;
}

/* ------------------------------------------------------------------------
 * Task 3 - Longest prefix match
 * --------------------------------------------------------------------- */

static bool is_space(char c)
{
  return c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\v' || c == '\f';
}

static const char *skip_space(const char *p)
{
  while (is_space(*p)) {
    p++;
  }
  return p;
}

/**
 * Copy the next whitespace-separated field from *p into buf, which holds size
 * bytes including the '\0', and advance *p past it. Returns false if there is
 * no field or it does not fit.
 */
static bool next_field(const char **p, char *buf, size_t size)
{
  const char *start = skip_space(*p);
  const char *end = start;
  while (*end != '\0' && !is_space(*end)) {
    end++;
  }
  size_t len = (size_t)(end - start);
  if (len == 0 || len >= size) {
    return false;
  }
  memcpy(buf, start, len);
  buf[len] = '\0';
  *p = end;
  return true;
}

bool route_parse(const char *line, route_t *route)
{
  if (line == NULL || route == NULL) {
    return false;
  }
  char dest[CIDR_STR_LEN];
  char gateway[IP_STR_LEN];
  char dev[DEV_NAME_LEN];
  const char *p = line;
  if (!next_field(&p, dest, sizeof dest) || !next_field(&p, gateway, sizeof gateway) ||
      !next_field(&p, dev, sizeof dev) || *skip_space(p) != '\0') {
    return false;
  }

  route_t result;
  memset(&result, 0, sizeof result);
  if (strcmp(dest, "default") == 0) {
    result.dest.addr = 0;
    result.dest.prefix = 0;
  } else if (!cidr_parse(dest, &result.dest) || !cidr_is_network(result.dest)) {
    return false;
  }
  if (strcmp(gateway, "direct") == 0) {
    result.gateway = 0;
  } else if (!ip_parse(gateway, &result.gateway)) {
    return false;
  }
  memcpy(result.dev, dev, sizeof dev);
  *route = result;
  return true;
}

fwd_table_t *table_create(void)
{
  fwd_table_t *table = calloc(1, sizeof *table);
  return table; // NULL if calloc() failed
}

void table_destroy(fwd_table_t *table)
{
  if (table == NULL) {
    return;
  }
  free(table->routes);
  free(table);
}

size_t table_count(const fwd_table_t *table)
{
  return table == NULL ? 0 : table->count;
}

bool table_add(fwd_table_t *table, const route_t *route)
{
  if (table == NULL || route == NULL) {
    return false;
  }
  if (route->dest.prefix > 32 || !cidr_is_network(route->dest)) {
    return false;
  }
  for (size_t i = 0; i < table->count; i++) {
    const cidr_t *have = &table->routes[i].dest;
    if (have->addr == route->dest.addr && have->prefix == route->dest.prefix) {
      return false;
    }
  }
  if (table->count == table->capacity) {
    size_t capacity = table->capacity == 0 ? 8 : table->capacity * 2;
    route_t *routes = realloc(table->routes, capacity * sizeof *routes);
    if (routes == NULL) { // GCOVR_EXCL_START
      return false;
    } // GCOVR_EXCL_STOP
    table->routes = routes;
    table->capacity = capacity;
  }
  route_t *slot = &table->routes[table->count];
  *slot = *route;
  slot->dev[DEV_NAME_LEN - 1] = '\0';
  table->count++;
  return true;
}

      *comment = '\0';
    }
    if (*skip_space(line) == '\0') {
      continue;
    }
    route_t route;
    if (!route_parse(line, &route) || !table_add(table, &route)) {
      if (bad_line != NULL) {
        *bad_line = number;
      }
      ok = false;
      break;
    }
  }
  if (ok && ferror(in)) {
    ok = false; // bad_line stays 0: the file itself could not be read
  }
  free(line);
  return ok;
}

const route_t *table_lookup(const fwd_table_t *table, uint32_t dest)
{
  if (table == NULL) {
    return NULL;
  }
  const route_t *best = NULL;
  for (size_t i = 0; i < table->count; i++) {
    const route_t *route = &table->routes[i];
    if (cidr_contains(route->dest, dest) &&
        (best == NULL || route->dest.prefix > best->dest.prefix)) {
      best = route;
    }
  }
  return best;
}

#ifndef LAB_H
#define LAB_H

#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>

/*
 * Every address in this file is a uint32_t in host byte order, so the first
 * octet of 192.168.1.10 is the most significant byte: 0xC0A8010A.
 */

/** Room for the longest dotted quad, "255.255.255.255", and its '\0'. */
#define IP_STR_LEN 16

/** Room for the longest CIDR string, "255.255.255.255/32", and its '\0'. */
#define CIDR_STR_LEN 19

/** Room for an interface name; IFNAMSIZ on Linux, including the '\0'. */
#define DEV_NAME_LEN 16

/** An address block in CIDR notation: an address and a prefix length. */
typedef struct {
  uint32_t addr;  /**< any address in the block, host byte order */
  uint8_t prefix; /**< 0 to 32 */
} cidr_t;

/** One entry in a forwarding table. */
typedef struct {
  cidr_t dest;            /**< destination block; host bits must be zero */
  uint32_t gateway;       /**< next hop, or 0 when the block is directly connected */
  char dev[DEV_NAME_LEN]; /**< outgoing interface, for example "eth0" */
} route_t;

/** A forwarding table. Its layout is private to lab.c. */
typedef struct fwd_table fwd_table_t;

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

/**
 * @brief Parse a dotted quad such as "192.168.1.10".
 *
 * Exactly four decimal octets from 0 to 255 separated by single periods, with
 * nothing before, between or after them. An octet with a leading zero ("010")
 * is rejected, because inet_aton() reads it as octal and that is never what a
 * person typing it meant.
 *
 * @param text The string to parse.
 * @param addr Where the address is stored on success; untouched on failure.
 * @return true if text is a valid dotted quad, false otherwise (including NULL).
 */
bool ip_parse(const char *text, uint32_t *addr);

/**
 * @brief Format an address as a dotted quad.
 * @param addr The address, host byte order.
 * @param out A buffer of at least IP_STR_LEN bytes.
 */
void ip_format(uint32_t addr, char out[IP_STR_LEN]);

/**
 * @brief Parse a block in CIDR notation such as "192.168.1.0/24".
 *
 * The address follows the rules of ip_parse() and the prefix is a decimal
 * number from 0 to 32 with no leading zero. The slash and the prefix are
 * required. Host bits may be set ("132.178.227.11/25" is an interface
 * address), so use cidr_is_network() when they must not be.
 *
 * @param text The string to parse.
 * @param block Where the block is stored on success; untouched on failure.
 * @return true if text is valid CIDR notation, false otherwise (including NULL).
 */
bool cidr_parse(const char *text, cidr_t *block);

/**
 * @brief Format a block as "a.b.c.d/p", using the address exactly as stored.
 *
 * A prefix over 32 is written as 32, the way cidr_mask() treats it.
 *
 * @param block The block.
 * @param out A buffer of at least CIDR_STR_LEN bytes.
 */
void cidr_format(cidr_t block, char out[CIDR_STR_LEN]);

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

/**
 * @brief The subnet mask for a prefix length: /24 is 255.255.255.0.
 *
 * Careful with /0. Shifting a 32 bit value by 32 is undefined behavior in C.
 *
 * @param prefix 0 to 32; anything larger is treated as 32.
 * @return The mask, host byte order.
 */
uint32_t cidr_mask(uint8_t prefix);

/** @brief The lowest address in the block: the address with the host bits cleared. */
uint32_t cidr_network(cidr_t block);

/** @brief The highest address in the block: the address with the host bits set. */
uint32_t cidr_broadcast(cidr_t block);

/**
 * @brief The first address that can be given to a host.
 *
 * For /0 to /30 that is the network address plus one. A /31 is a point to
 * point link (RFC 3021) where both addresses are usable, and a /32 is a single
 * host, so for those two the first host is the network address.
 */
uint32_t cidr_first_host(cidr_t block);

/**
 * @brief The last address that can be given to a host.
 *
 * The broadcast address minus one for /0 to /30, and the broadcast address
 * itself for /31 and /32 (see cidr_first_host()).
 */
uint32_t cidr_last_host(cidr_t block);

/**
 * @brief How many addresses in the block can be given to hosts.
 *
 * 2^(32 - prefix) - 2 for /0 to /30, 2 for /31, and 1 for /32. For a /0 the
 * 2^32 in that formula needs 33 bits, more than a uint32_t holds, which is why
 * this returns a uint64_t.
 */
uint64_t cidr_host_count(cidr_t block);

/** @brief true if addr falls inside the block. */
bool cidr_contains(cidr_t block, uint32_t addr);

/** @brief true if the block's host bits are all zero, so it names a network. */
bool cidr_is_network(cidr_t block);

/**
 * @brief The i-th of n equally sized subnets of a block.
 *
 * Splitting a /24 into 4 gives four /26 blocks, and subnet 0 is the lowest.
 * Equal sizes are only possible when n is a power of two, and the new prefix
 * can not pass 32.
 *
 * @param block The block to split; it must be a network (cidr_is_network()).
 * @param n How many subnets, a power of two from 1 up.
 * @param i Which subnet, 0 to n - 1.
 * @param out Where the subnet is stored on success; untouched on failure.
 * @return false if out is NULL, the block is not a network, n is not a power
 *         of two, the split would need a prefix longer than 32, or i is out of
 *         range.
 */
bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out);

/* ------------------------------------------------------------------------
 * Task 3 - Longest prefix match
 * --------------------------------------------------------------------- */

/**
 * @brief Parse one line of a forwarding table file.
 *
 * A line is three fields separated by spaces or tabs:
 *
 *     <destination> <gateway> <device>
 *
 * The destination is a block in CIDR notation with no host bits set, or the
 * word "default" for 0.0.0.0/0. The gateway is a dotted quad, or the word
 * "direct" for a directly connected block (stored as 0). The device is 1 to
 * DEV_NAME_LEN - 1 characters. Leading and trailing whitespace, including a
 * trailing newline, is ignored. Comments are not handled here.
 *
 * @param line The line.
 * @param route Where the route is stored on success; untouched on failure.
 * @return true if the line is a valid route, false otherwise (including NULL).
 */
bool route_parse(const char *line, route_t *route);

/**
 * @brief Create an empty forwarding table.
 * @return The table, which the caller frees with table_destroy(), or NULL if
 *         memory could not be allocated.
 */
fwd_table_t *table_create(void);

/** @brief Free a table and everything in it. NULL is allowed and does nothing. */
void table_destroy(fwd_table_t *table);

/** @brief The number of routes in the table; 0 for NULL. */
size_t table_count(const fwd_table_t *table);

/**
 * @brief Add a copy of a route to the table, growing it as needed.
 * @return false if either argument is NULL, the destination has host bits
 *         set, the table already holds a route to the same block, or memory
 *         could not be allocated. The table is unchanged on failure.
 */
bool table_add(fwd_table_t *table, const route_t *route);

/**
 * @brief Read a forwarding table file and add every route in it.
 *
 * Everything from a '#' to the end of a line is a comment, and a line that is
 * blank once the comment is removed is skipped. Every other line must be a
 * valid route (route_parse()) that table_add() accepts.
 *
 * @param table The table to add to.
 * @param in The open file to read.
 * @param bad_line If not NULL, set to the 1-based number of the first line
 *        that failed, or 0 if the arguments were NULL or the file could not
 *        be read.
 * @return true if every line loaded. On false, routes from the lines before
 *         the bad one stay in the table.
 */
bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line);

/**
 * @brief Longest prefix match: the route whose block contains dest and has the
 *        longest prefix.
 *
 * A default route (0.0.0.0/0) matches everything, so it is chosen only when
 * nothing more specific does. The order the routes were added in does not
 * matter.
 *
 * @return The matching route, which stays owned by the table, or NULL if no
 *         route matches (or table is NULL).
 */
const route_t *table_lookup(const fwd_table_t *table, uint32_t dest);

#endif // LAB_H

#include "lab.h"
#include <errno.h>
#include <inttypes.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#ifdef TEST
#define main main_exclude
#endif

/** Exit codes. */
enum {
  EXIT_OK = 0,
  EXIT_USAGE = 1,       /* bad command line or malformed address */
  EXIT_TABLE = 2,       /* table file can not be opened or has a bad line */
  EXIT_UNREACHABLE = 3, /* an address matched no route */
};

static void usage(FILE *out)
{
  fprintf(out, "Usage: myapp info <address>/<prefix>\n"
               "       myapp split <network>/<prefix> <count>\n"
               "       myapp lookup <table-file> <address>...\n");
}

static int cmd_info(const char *arg)
{
  cidr_t block;
  if (!cidr_parse(arg, &block)) {
    fprintf(stderr, "myapp: '%s' is not a valid <address>/<prefix>\n", arg);
    return EXIT_USAGE;
  }
  char buf[IP_STR_LEN];
  ip_format(block.addr, buf);
  printf("address    %s\n", buf);
  printf("prefix     /%u\n", (unsigned)block.prefix);
  ip_format(cidr_mask(block.prefix), buf);
  printf("netmask    %s\n", buf);
  ip_format(cidr_network(block), buf);
  printf("network    %s\n", buf);
  ip_format(cidr_broadcast(block), buf);
  printf("broadcast  %s\n", buf);
  ip_format(cidr_first_host(block), buf);
  printf("first host %s\n", buf);
  ip_format(cidr_last_host(block), buf);
  printf("last host  %s\n", buf);
  printf("hosts      %" PRIu64 "\n", cidr_host_count(block));
  return EXIT_OK;
}

/** Parse a subnet count: decimal digits only, no sign, no leading zero. */
static bool parse_count(const char *text, uint32_t *count)
{
  if (text[0] < '1' || text[0] > '9') {
    return false;
  }
  errno = 0;
  char *end;
  unsigned long value = strtoul(text, &end, 10);
  if (*end != '\0' || errno == ERANGE || value > UINT32_MAX) {
    return false;
  }
  *count = (uint32_t)value;
  return true;
}

static int cmd_split(const char *block_arg, const char *count_arg)
{
  cidr_t block;
  if (!cidr_parse(block_arg, &block)) {
    fprintf(stderr, "myapp: '%s' is not a valid <network>/<prefix>\n", block_arg);
    return EXIT_USAGE;
  }
  if (!cidr_is_network(block)) {
    char network[CIDR_STR_LEN];
    cidr_t fixed = {cidr_network(block), block.prefix};
    cidr_format(fixed, network);
    fprintf(stderr, "myapp: '%s' has host bits set; the network is %s\n", block_arg,
            network);
    return EXIT_USAGE;
  }
  uint32_t count;
  if (!parse_count(count_arg, &count) || (count & (count - 1)) != 0) {
    fprintf(stderr, "myapp: '%s' is not a power of two\n", count_arg);
    return EXIT_USAGE;
  }
  cidr_t subnet;
  if (!cidr_subnet(block, count, 0, &subnet)) {
    fprintf(stderr, "myapp: can not split a /%u into %s subnets, the prefix would pass /32\n",
            (unsigned)block.prefix, count_arg);
    return EXIT_USAGE;
  }
  char buf[CIDR_STR_LEN];
  for (uint32_t i = 0; i < count; i++) {
    cidr_subnet(block, count, i, &subnet);
    cidr_format(subnet, buf);
    printf("%s\n", buf);
  }
  return EXIT_OK;
}

static int cmd_lookup(const char *path, int naddrs, char **addrs)
{
  FILE *in = fopen(path, "r");
  if (in == NULL) {
    fprintf(stderr, "myapp: %s: %s\n", path, strerror(errno));
    return EXIT_TABLE;
  }
  fwd_table_t *table = table_create();
  if (table == NULL) {
    fprintf(stderr, "myapp: out of memory\n");
    fclose(in);
    return EXIT_TABLE;
  }
  size_t bad_line;
  bool loaded = table_load(table, in, &bad_line);
  fclose(in);
  if (!loaded) {
    if (bad_line == 0) {
      fprintf(stderr, "myapp: %s: could not be read\n", path);
    } else {
      fprintf(stderr, "myapp: %s:%zu: not a valid route\n", path, bad_line);
    }
    table_destroy(table);
    return EXIT_TABLE;
  }

  bool malformed = false;
  bool unreachable = false;
  for (int i = 0; i < naddrs; i++) {
    uint32_t dest;
    if (!ip_parse(addrs[i], &dest)) {
      fprintf(stderr, "myapp: '%s' is not a valid address\n", addrs[i]);
      malformed = true;
      continue;
    }
    char buf[IP_STR_LEN];
    ip_format(dest, buf);
    const route_t *route = table_lookup(table, dest);
    if (route == NULL) {
      printf("%s unreachable\n", buf);
      unreachable = true;
    } else if (route->gateway != 0) {
      char gw[IP_STR_LEN];
      ip_format(route->gateway, gw);
      printf("%s via %s dev %s\n", buf, gw, route->dev);
    } else {
      printf("%s dev %s\n", buf, route->dev);
    }
  }
  table_destroy(table);

  if (malformed) {
    return EXIT_USAGE;
  }
  return unreachable ? EXIT_UNREACHABLE : EXIT_OK;
}

int main(int argc, char **argv)
{
  if (argc == 1) {
    usage(stdout);
    return EXIT_OK;
  }
  if (strcmp(argv[1], "info") == 0 && argc == 3) {
    return cmd_info(argv[2]);
  }
  if (strcmp(argv[1], "split") == 0 && argc == 4) {
    return cmd_split(argv[2], argv[3]);
  }
  if (strcmp(argv[1], "lookup") == 0 && argc >= 4) {
    return cmd_lookup(argv[2], argc - 3, argv + 3);
  }
  usage(stderr);
  return EXIT_USAGE;
}

void test_cidr_subnet_edges(void)
{
  cidr_t out;
  // n = 1 hands back the block itself, even for /0 and /32.
  TEST_ASSERT_TRUE(cidr_subnet(block(0, 0), 1, 0, &out));
  TEST_ASSERT_EQUAL_HEX32(0, out.addr);
  TEST_ASSERT_EQUAL_UINT8(0, out.prefix);
  TEST_ASSERT_TRUE(cidr_subnet(block(0xFFFFFFFF, 32), 1, 0, &out));
  TEST_ASSERT_EQUAL_HEX32(0xFFFFFFFF, out.addr);
  TEST_ASSERT_EQUAL_UINT8(32, out.prefix);

  // 0.0.0.0/0 in two halves.
  TEST_ASSERT_TRUE(cidr_subnet(block(0, 0), 2, 1, &out));
  TEST_ASSERT_EQUAL_HEX32(0x80000000, out.addr);
  TEST_ASSERT_EQUAL_UINT8(1, out.prefix);

  // 0.0.0.0/0 into 2^31 pieces: the last one is a /31 at the very top.
  TEST_ASSERT_TRUE(cidr_subnet(block(0, 0), 0x80000000u, 0x7FFFFFFFu, &out));
  TEST_ASSERT_EQUAL_HEX32(0xFFFFFFFE, out.addr);
  TEST_ASSERT_EQUAL_UINT8(31, out.prefix);

  // A /30 into four /32s.
  TEST_ASSERT_TRUE(cidr_subnet(block(IP(10, 0, 0, 4), 30), 4, 3, &out));
  TEST_ASSERT_EQUAL_HEX32(IP(10, 0, 0, 7), out.addr);
  TEST_ASSERT_EQUAL_UINT8(32, out.prefix);

  // A /31 into two /32s.
  TEST_ASSERT_TRUE(cidr_subnet(block(IP(10, 0, 0, 6), 31), 2, 1, &out));
  TEST_ASSERT_EQUAL_HEX32(IP(10, 0, 0, 7), out.addr);
  TEST_ASSERT_EQUAL_UINT8(32, out.prefix);
}

void test_cidr_subnet_rejects(void)
{
  cidr_t net = block(IP(192, 168, 1, 0), 24);
  cidr_t out = {0xDEADBEEF, 99};
  TEST_ASSERT_FALSE(cidr_subnet(net, 4, 0, NULL));
  TEST_ASSERT_FALSE(cidr_subnet(block(IP(192, 168, 1, 5), 24), 4, 0, &out)); // host bits
  TEST_ASSERT_FALSE(cidr_subnet(net, 0, 0, &out));                          // n = 0
  TEST_ASSERT_FALSE(cidr_subnet(net, 3, 0, &out));                          // not 2^k
  TEST_ASSERT_FALSE(cidr_subnet(net, 6, 0, &out));
  TEST_ASSERT_FALSE(cidr_subnet(net, 512, 0, &out));                        // past /32
  TEST_ASSERT_FALSE(cidr_subnet(block(0xFFFFFFFF, 32), 2, 0, &out));
  TEST_ASSERT_FALSE(cidr_subnet(net, 4, 4, &out));                          // i out of range
  TEST_ASSERT_FALSE(cidr_subnet(net, 1, 1, &out));
  TEST_ASSERT_EQUAL_HEX32(0xDEADBEEF, out.addr);
  TEST_ASSERT_EQUAL_UINT8(99, out.prefix);
  TEST_ASSERT_TRUE(cidr_subnet(net, 256, 255, &out)); // exactly /32 is fine
  TEST_ASSERT_EQUAL_HEX32(IP(192, 168, 1, 255), out.addr);
  TEST_ASSERT_EQUAL_UINT8(32, out.prefix);
}

void test_route_parse_valid(void)
{
  route_t r;
  TEST_ASSERT_TRUE(route_parse("10.1.2.128/25 10.1.2.1 eth2", &r));
  TEST_ASSERT_EQUAL_HEX32(IP(10, 1, 2, 128), r.dest.addr);
  TEST_ASSERT_EQUAL_UINT8(25, r.dest.prefix);
  TEST_ASSERT_EQUAL_HEX32(IP(10, 1, 2, 1), r.gateway);
  TEST_ASSERT_EQUAL_STRING("eth2", r.dev);

  TEST_ASSERT_TRUE(route_parse("  default\t132.178.227.1   eno1  \r\n", &r));
  TEST_ASSERT_EQUAL_HEX32(0, r.dest.addr);
  TEST_ASSERT_EQUAL_UINT8(0, r.dest.prefix);
  TEST_ASSERT_EQUAL_HEX32(IP(132, 178, 227, 1), r.gateway);
  TEST_ASSERT_EQUAL_STRING("eno1", r.dev);

  TEST_ASSERT_TRUE(route_parse("\v10.0.0.0/8\fdirect eth1\f\v", &r));
  TEST_ASSERT_EQUAL_STRING("eth1", r.dev);

  TEST_ASSERT_TRUE(route_parse("132.178.227.0/25 direct eno1\n", &r));
  TEST_ASSERT_EQUAL_HEX32(0, r.gateway);

  TEST_ASSERT_TRUE(route_parse("0.0.0.0/0 direct abcdefghijklmno", &r)); // 15 chars
  TEST_ASSERT_EQUAL_STRING("abcdefghijklmno", r.dev);

  TEST_ASSERT_TRUE(route_parse("255.255.255.255/32 0.0.0.0 x", &r));
  TEST_ASSERT_EQUAL_HEX32(0xFFFFFFFF, r.dest.addr);
  TEST_ASSERT_EQUAL_UINT8(32, r.dest.prefix);
}

  for (size_t i = 0; i < sizeof bad / sizeof bad[0]; i++) {

}

static const route_t nested[] = {
    {{0x0A000000, 8}, 0x0A010001, "eth1"},  // 10.0.0.0/8
    {{0x0A010000, 16}, 0, "eth1"},          // 10.1.0.0/16
    {{0x0A010200, 24}, 0, "eth2"},          // 10.1.2.0/24
    {{0x0A010280, 25}, 0x0A010201, "eth2"}, // 10.1.2.128/25
};

static void check_nested(const fwd_table_t *t)
{
  const route_t *r = table_lookup(t, IP(10, 1, 2, 200));
  TEST_ASSERT_NOT_NULL(r);
  TEST_ASSERT_EQUAL_UINT8(25, r->dest.prefix);
  r = table_lookup(t, IP(10, 1, 2, 5));
  TEST_ASSERT_NOT_NULL(r);
  TEST_ASSERT_EQUAL_UINT8(24, r->dest.prefix);
  r = table_lookup(t, IP(10, 1, 9, 9));
  TEST_ASSERT_NOT_NULL(r);
  TEST_ASSERT_EQUAL_UINT8(16, r->dest.prefix);
  r = table_lookup(t, IP(10, 2, 0, 1));
  TEST_ASSERT_NOT_NULL(r);
  TEST_ASSERT_EQUAL_UINT8(8, r->dest.prefix);
  // No default route, so anything outside 10/8 is unreachable.
  TEST_ASSERT_NULL(table_lookup(t, IP(11, 0, 0, 0)));
  TEST_ASSERT_NULL(table_lookup(t, IP(9, 255, 255, 255)));
  TEST_ASSERT_NULL(table_lookup(t, 0));
  TEST_ASSERT_NULL(table_lookup(t, 0xFFFFFFFF));
}

void test_lookup_longest_wins_any_order(void)
{
  const size_t orders[][4] = {
      {0, 1, 2, 3}, {3, 2, 1, 0}, {1, 3, 0, 2}, {2, 0, 3, 1},
  };
  for (size_t o = 0; o < sizeof orders / sizeof orders[0]; o++) {
    fwd_table_t *t = table_create();
    for (size_t i = 0; i < 4; i++) {
      TEST_ASSERT_TRUE(table_add(t, &nested[orders[o][i]]));
    }
    check_nested(t);
    table_destroy(t);
  }
}
